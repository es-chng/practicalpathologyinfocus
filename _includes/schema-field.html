{%- comment -%} Renders one block according to its declared shape. shape: repeat -> iterate the value as a list of objects, rendering each sub-field through schema-field.html shape: block -> render each named sub-field once through schema-field.html otherwise -> leaf field; hand to schema-field.html {%- endcomment -%} {%- assign b = include.block -%} {%- assign name = include.name -%} {%- assign val = include.value -%} {%- if b.shape == "repeat" -%} {%- for item in val -%}
{%- for sub_name in b.item_order -%} {%- assign sub_def = b.item[sub_name] -%} {%- assign sub_value = item[sub_name] -%} {% include schema-field.html f=sub_def v=sub_value %} {%- endfor -%}
{%- endfor -%} {%- elsif b.shape == "block" -%}
{%- for sub_name in b.field_order -%} {%- assign sub_def = b.fields[sub_name] -%} {%- assign sub_value = val[sub_name] -%} {% include schema-field.html f=sub_def v=sub_value %} {%- endfor -%}
{%- else -%} {% include schema-field.html f=b v=val %} {%- endif -%}
