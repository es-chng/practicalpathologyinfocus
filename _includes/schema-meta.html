{%- comment -%} Discovers teaser and badge keys from an article schema. Supports: - current structure/blocks format (block name = article key) - legacy fields[] format (f.key) Sets in the caller scope: schema_teaser_key schema_badge_key {%- endcomment -%} {%- assign schema_teaser_key = nil -%} {%- assign schema_badge_key = nil -%} {%- assign _schema = include.schema -%} {%- if _schema -%} {%- if _schema.blocks -%} {%- if _schema.structure -%} {%- for _bname in _schema.structure -%} {%- assign _bdef = _schema.blocks[_bname] -%} {%- if _bdef.teaser -%}{%- assign schema_teaser_key = _bname -%}{%- endif -%} {%- if _bdef.style == "badge" -%}{%- assign schema_badge_key = _bname -%}{%- endif -%} {%- endfor -%} {%- else -%} {%- comment -%} fallback: iterate block map values with name from structure only {%- endcomment -%} {%- for _bname in _schema.blocks -%} {%- assign _bdef = _schema.blocks[_bname] -%} {%- if _bdef.teaser -%}{%- assign schema_teaser_key = _bname -%}{%- endif -%} {%- if _bdef.style == "badge" -%}{%- assign schema_badge_key = _bname -%}{%- endif -%} {%- endfor -%} {%- endif -%} {%- elsif _schema.fields -%} {%- for f in _schema.fields -%} {%- if f.teaser -%}{%- assign schema_teaser_key = f.key -%}{%- endif -%} {%- if f.style == "badge" -%}{%- assign schema_badge_key = f.key -%}{%- endif -%} {%- endfor -%} {%- endif -%} {%- endif -%}
